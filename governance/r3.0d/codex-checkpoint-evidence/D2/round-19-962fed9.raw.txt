  //   'plain-object' — v is a non-null, non-array object whose prototype is exactly
  //                    Object.prototype or null. Class instances are rejected.

  // deepOriginalShapeAudit(v, depth) — recursive structural validator. Returns true ONLY if v
  // is composed exclusively of plain objects (Object.prototype proto), plain arrays
  // (Array.prototype proto, NOT subclass), strings, finite numbers, booleans, null. Rejects:
  //   - class instances (non-Object.prototype proto)
  //   - Array subclasses (non-Array.prototype proto)
  //   - Proxies that lie about their prototype (lying still produces a wrong proto here)
  //   - Symbol-keyed own properties at any depth
  //   - Non-enumerable own properties at any depth
  //   - Accessor descriptors at any depth
  //   - Function values
  //   - Infinity / NaN numbers
  //   - Recursion past depth cap (32)
  // Calls captured intrinsics only; the audit itself never invokes a [[Get]] on the input
  // (uses descriptor reads exclusively).
  function deepOriginalShapeAudit(v, depth) {
    if (depth === undefined) depth = 0;
    if (depth > 32) return false;
    if (v === null) return true;
    var t = typeof v;
    if (t === 'string' || t === 'boolean') return true;
    if (t === 'number') return safeNumberIsFinite(v);
    if (t === 'function' || t === 'symbol' || t === 'undefined') return false;
    if (t !== 'object') return false;
    // prototype check
    var proto = safeGetPrototypeOf(v);
    var isPlainArray = false;
    if (proto === null) {
      // Object.create(null) — accept; treated as plain object with null proto
    } else if (proto === _ObjectPrototype) {
      // plain object
    } else if (proto === _ArrayPrototype) {
      // plain array — additional check: Array.isArray must agree
      var isArr = _apply(_ArrayIsArray, _Array, [v]);
      if (isArr !== true) return false;
      isPlainArray = true;
    } else {
      return false; // class instance / Array subclass / mutated prototype
    }
    // structural key audit (own keys, including non-enumerable, including Symbol)
    var allKeys = safeOwnKeys(v);
    if (allKeys === null) return false;
    var enumKeysOnly = safeKeys(v);
    if (enumKeysOnly === null) return false;
    // For plain Array: an extra non-enumerable own property "length" is intrinsic (every
    // Array has it by spec). Accept exactly that one extra. Any OTHER non-enumerable or
    // Symbol key is a rejection signal.
    var allowedExtra = isPlainArray ? 1 : 0;
    if (allKeys.length !== enumKeysOnly.length + allowedExtra) return false;
    var len = allKeys.length;
    for (var i = 0; i < len; i++) {
      var k = allKeys[i];
      if (typeof k === 'symbol') return false;
      var d = safeGetOwnDescriptor(v, k);
      if (d === null || d === undefined) return false;
      if (!('value' in d)) return false; // accessor descriptor
      // Allow the intrinsic Array length (non-enumerable). For all other keys, must be enumerable.
      if (d.enumerable !== true) {
        if (!(isPlainArray && k === 'length')) return false;
      }
      // Recurse into nested
      var child = d.value;
      if (typeof child === 'object' && child !== null) {
        if (!deepOriginalShapeAudit(child, depth + 1)) return false;
      }
      else if (typeof child === 'number') {
        if (!safeNumberIsFinite(child)) return false;
      }
      else if (typeof child === 'function' || typeof child === 'symbol' || typeof child === 'undefined') {
        return false;
      }
    }
    return true;
  }

// These are used by the runRebind harness to install + restore the hostile binding without

const _origObjectDefineProperty = Object.defineProperty;

let pass = 0, fail = 0;
function chk(name, cond, detail) {

  else {

// runRebind(container, key, hostileImpl, action)
//   container: the object to rebind on (e.g. Array.prototype, Object, Reflect)
//   key:       the property name to rebind
//   hostileImpl: function to install (its body counts hits + may self-restore)
//   action:    () => result — the wrapper invocation to test under the rebind
// Returns: { result, hits } where hits is the count of hostile invocations.
//
// IMPORTANT: install + restore use the captured _origObjectDefineProperty via captured
// _origReflectApply so the harness itself never invokes the (potentially-rebound) ambient
// Object.defineProperty. Otherwise rebinding Object.defineProperty would cause the harness
// to invoke its own hostile during install/restore, corrupting subsequent tests.
function runRebind(container, key, hostileImpl, action) {
  const orig = container[key];
  let hits = 0;
  const hostile = function () {
    hits++;
    return hostileImpl.apply(this, arguments);
  };
  let result;
  try {
    _origReflectApply(_origObjectDefineProperty, Object, [container, key, { value: hostile, writable: true, enumerable: true, configurable: true }]);
    result = action();
  } finally {
    try { _origReflectApply(_origObjectDefineProperty, Object, [container, key, { value: orig, writable: true, enumerable: true, configurable: true }]); }
    catch (e) { container[key] = orig; }
  }
  return { result: result, hits: hits };
}

   445	  // is composed exclusively of plain objects (Object.prototype proto), plain arrays

   447	  //   - class instances (non-Object.prototype proto)

   450	  //   - Symbol-keyed own properties at any depth

   453	  //   - Function values
   454	  //   - Infinity / NaN numbers

   457	  // (uses descriptor reads exclusively).

   459	    if (depth === undefined) depth = 0;

   461	    if (v === null) return true;
   462	    var t = typeof v;
   463	    if (t === 'string' || t === 'boolean') return true;
   464	    if (t === 'number') return safeNumberIsFinite(v);
   465	    if (t === 'function' || t === 'symbol' || t === 'undefined') return false;
   466	    if (t !== 'object') return false;

   468	    var proto = safeGetPrototypeOf(v);
   469	    var isPlainArray = false;
   470	    if (proto === null) {
   471	      // Object.create(null) — accept; treated as plain object with null proto
   472	    } else if (proto === _ObjectPrototype) {
   473	      // plain object
   474	    } else if (proto === _ArrayPrototype) {
   475	      // plain array — additional check: Array.isArray must agree
   476	      var isArr = _apply(_ArrayIsArray, _Array, [v]);
   477	      if (isArr !== true) return false;
   478	      isPlainArray = true;
   479	    } else {
   480	      return false; // class instance / Array subclass / mutated prototype
   481	    }
   482	    // structural key audit (own keys, including non-enumerable, including Symbol)
   483	    var allKeys = safeOwnKeys(v);
   484	    if (allKeys === null) return false;
   485	    var enumKeysOnly = safeKeys(v);
   486	    if (enumKeysOnly === null) return false;
   487	    // For plain Array: an extra non-enumerable own property "length" is intrinsic (every
   488	    // Array has it by spec). Accept exactly that one extra. Any OTHER non-enumerable or
   489	    // Symbol key is a rejection signal.
   490	    var allowedExtra = isPlainArray ? 1 : 0;
   491	    if (allKeys.length !== enumKeysOnly.length + allowedExtra) return false;
   492	    var len = allKeys.length;
   493	    for (var i = 0; i < len; i++) {
   494	      var k = allKeys[i];

   496	      var d = safeGetOwnDescriptor(v, k);
   497	      if (d === null || d === undefined) return false;
   498	      if (!('value' in d)) return false; // accessor descriptor
   499	      // Allow the intrinsic Array length (non-enumerable). For all other keys, must be enumerable.
   500	      if (d.enumerable !== true) {
   501	        if (!(isPlainArray && k === 'length')) return false;
   502	      }

   504	      var child = d.value;
   505	      if (typeof child === 'object' && child !== null) {
   506	        if (!deepOriginalShapeAudit(child, depth + 1)) return false;
   507	      }
   508	      else if (typeof child === 'number') {
   509	        if (!safeNumberIsFinite(child)) return false;
   510	      }
   511	      else if (typeof child === 'function' || typeof child === 'symbol' || typeof child === 'undefined') {
   512	        return false;
   513	      }
   514	    }
   515	    return true;
   516	  }

    29	// depending on the (potentially-rebound) ambient Object.defineProperty.
    30	const _origObjectDefineProperty = Object.defineProperty;
    31	const _origReflectApply = Reflect.apply;

    33	let pass = 0, fail = 0;
    34	function chk(name, cond, detail) {
    35	  if (cond) pass++;
    36	  else {
    37	    fail++;
    38	    console.log('  ✗ ' + name + (detail !== undefined ? '  ' + (typeof detail === 'string' ? detail : JSON.stringify(detail)) : ''));
    39	  }

    42	// runRebind(container, key, hostileImpl, action)
    43	//   container: the object to rebind on (e.g. Array.prototype, Object, Reflect)
    44	//   key:       the property name to rebind
    45	//   hostileImpl: function to install (its body counts hits + may self-restore)
    46	//   action:    () => result — the wrapper invocation to test under the rebind
    47	// Returns: { result, hits } where hits is the count of hostile invocations.
    48	//
    49	// IMPORTANT: install + restore use the captured _origObjectDefineProperty via captured
    50	// _origReflectApply so the harness itself never invokes the (potentially-rebound) ambient
    51	// Object.defineProperty. Otherwise rebinding Object.defineProperty would cause the harness
    52	// to invoke its own hostile during install/restore, corrupting subsequent tests.
    53	function runRebind(container, key, hostileImpl, action) {
    54	  const orig = container[key];
    55	  let hits = 0;
    56	  const hostile = function () {
    57	    hits++;
    58	    return hostileImpl.apply(this, arguments);
    59	  };
    60	  let result;
    61	  try {
    62	    _origReflectApply(_origObjectDefineProperty, Object, [container, key, { value: hostile, writable: true, enumerable: true, configurable: true }]);
    63	    result = action();
    64	  } finally {
    65	    try { _origReflectApply(_origObjectDefineProperty, Object, [container, key, { value: orig, writable: true, enumerable: true, configurable: true }]); }
    66	    catch (e) { container[key] = orig; }
    67	  }
    68	  return { result: result, hits: hits };
    69	}
