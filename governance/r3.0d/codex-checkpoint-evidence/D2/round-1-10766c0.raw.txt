   239	  // DFS cycle detection over the directed edge list. Returns null if no cycle, or a representative
   240	  // node id participating in the cycle. White/grey/black colouring.
   241	  function _findCycle(nodeIds, adjacency) {
   242	    var WHITE = 0, GREY = 1, BLACK = 2;
   243	    var color = {};
   244	    for (var i = 0; i < nodeIds.length; i++) color[nodeIds[i]] = WHITE;
   245	    var stack = [];
   246	    function visit(u) {
   247	      color[u] = GREY;
   248	      stack.push(u);
   249	      var nbrs = adjacency[u] || [];

   251	        var v = nbrs[k];
   252	        if (color[v] === GREY) return v;
   253	        if (color[v] === WHITE) {
   254	          var c = visit(v);
   255	          if (c !== null) return c;
   256	        }
   257	      }
   258	      color[u] = BLACK;
   259	      stack.pop();
   260	      return null;
   261	    }
   262	    // Iterate in deterministic order (caller passes sorted nodeIds).
   263	    for (var j = 0; j < nodeIds.length; j++) {
   264	      var n = nodeIds[j];
   265	      if (color[n] === WHITE) {
   266	        var hit = visit(n);
   267	        if (hit !== null) return hit;
   268	      }
   269	    }
   270	    return null;
   271	  }

   273	  // Kahn's algorithm — deterministic by tie-breaking on sorted nodeId. Assumes no cycle (callers
   274	  // must check via _findCycle first).
   275	  function _topologicalOrder(nodeIds, adjacency) {
   276	    var inDegree = {};
   277	    var i;
   278	    for (i = 0; i < nodeIds.length; i++) inDegree[nodeIds[i]] = 0;
   279	    for (i = 0; i < nodeIds.length; i++) {
   280	      var nbrs = adjacency[nodeIds[i]] || [];
   281	      for (var k = 0; k < nbrs.length; k++) {
   282	        if (Object.prototype.hasOwnProperty.call(inDegree, nbrs[k])) inDegree[nbrs[k]]++;
   283	      }
   284	    }
   285	    var queue = [];
   286	    for (i = 0; i < nodeIds.length; i++) if (inDegree[nodeIds[i]] === 0) queue.push(nodeIds[i]);
   287	    queue.sort(_strcmp);
   288	    var out = [];
   289	    while (queue.length > 0) {
   290	      var u = queue.shift();
   291	      out.push(u);
   292	      var nbrs2 = (adjacency[u] || []).slice().sort(_strcmp);
   293	      for (var m = 0; m < nbrs2.length; m++) {
   294	        var v = nbrs2[m];
   295	        if (Object.prototype.hasOwnProperty.call(inDegree, v)) {
   296	          inDegree[v]--;
   297	          if (inDegree[v] === 0) {
   298	            // insert sorted
   299	            var lo = 0, hi = queue.length;
   300	            while (lo < hi) { var mid = (lo + hi) >>> 1; if (queue[mid] < v) lo = mid + 1; else hi = mid; }
   301	            queue.splice(lo, 0, v);
   302	          }
   303	        }
   304	      }
   305	    }
   306	    return out;
   307	  }

    chore(r3.0): state-aware production-consumer authorization in check-r3-phase-no-consumer.js
    
    The per-phase no-runtime-consumer validator was originally written for the contracts-only
    checkpoint (D1 / E0 / F0) and unconditionally rejected any renderer/js consumer of contracts/
    <phase>/. That posture is correct at the bootstrap floor but contradicts the surrounding
    governance contract once a production checkpoint authorizes a specific module to consume the
    contracts (state.json :: authorizedProductionPaths).
    
    This commit teaches the validator to read state.authorizedProductionPaths and emit
    PROD_UNAUTHORIZED_CONSUMER ONLY for consumers that are NOT on the authorized list. Authorized
    consumers PASS. The strictest fail-closed posture is preserved at the bootstrap floor (empty
    allow set yields any consumer flagged).
    
    This change is intentionally a STRUCTURAL ENHANCEMENT, never an across-the-board weakening:
    the new behaviour is exactly the old behaviour when the allow set is empty, and is strictly
    the surrounding governance's own authorization rule when the allow set is non-empty. The
    companion D2_EVIDENCE_GRAPH advance (separate commit) is the first user of the authorized
    allow list.
