    49	// Stack-based lexer. Template literal text is skipped; `${` opens an expression that is lexed inline
    50	// through this same loop; the matching `}` (tracked by brace depth, never confused by a `}` inside a
    51	// comment/string/regex within the expression) closes it back to template-literal mode.
    52	function tokenize(src) {
    53	  const T = []; let i = 0; const n = src.length;
    54	  const last = () => (T.length ? T[T.length - 1] : null);
    55	  const tplBraceStack = []; let braceDepth = 0; let mode = 'code';
    56	  while (i < n) {
    57	    if (mode === 'tpl') {
    58	      const c = src[i];
    59	      if (c === '\\') { i += 2; continue; }
    60	      if (c === '`') { i++; mode = 'code'; continue; }
    61	      if (c === '$' && src[i + 1] === '{') { T.push({ t: 'punct', v: '${', i }); i += 2; tplBraceStack.push(braceDepth); braceDepth++; mode = 'code'; continue; }
    62	      i++; continue;
    63	    }
    64	    const c = src[i];
    65	    if (c === ' ' || c === '\t' || c === '\r' || c === '\n') { i++; continue; }
    66	    if (c === '/' && src[i + 1] === '/') { i += 2; while (i < n && src[i] !== '\n') i++; continue; }
    67	    if (c === '/' && src[i + 1] === '*') { i += 2; while (i < n && !(src[i] === '*' && src[i + 1] === '/')) i++; i += 2; continue; }
    68	    if (c === '"' || c === "'") { let j = i + 1, v = ''; while (j < n) { if (src[j] === '\\') { v += (src[j + 1] || ''); j += 2; continue; } if (src[j] === c) break; v += src[j]; j++; } T.push({ t: 'str', v, i }); i = j + 1; continue; }
    69	    if (c === '`') { T.push({ t: 'tmpl', i }); i++; mode = 'tpl'; continue; }
    70	    if (c === '/') {
    71	      const l = last();
    72	      const division = l && ((l.t === 'word' && !KEYWORDS.has(l.v)) || l.t === 'num' || l.t === 'str' || l.t === 'tmpl' || l.t === 'regex' || (l.t === 'punct' && (l.v === ')' || l.v === ']')));
    73	      if (!division) { let j = i + 1, cls = false; while (j < n) { const e = src[j]; if (e === '\\') { j += 2; continue; } if (e === '[') cls = true; else if (e === ']') cls = false; else if (e === '/' && !cls) break; else if (e === '\n') break; j++; } j++; while (j < n && /[a-z]/i.test(src[j])) j++; T.push({ t: 'regex', i }); i = j; continue; }
    74	      T.push({ t: 'punct', v: '/', i }); i++; continue;
    75	    }
    76	    if (c === '{') { braceDepth++; T.push({ t: 'punct', v: '{', i }); i++; continue; }
    77	    if (c === '}') {
    78	      if (tplBraceStack.length && braceDepth === tplBraceStack[tplBraceStack.length - 1] + 1) { tplBraceStack.pop(); braceDepth--; T.push({ t: 'punct', v: '}', i }); i++; mode = 'tpl'; continue; }
    79	      braceDepth--; T.push({ t: 'punct', v: '}', i }); i++; continue;
    80	    }
    81	    if (/[A-Za-z_$]/.test(c)) { let j = i; while (j < n && /[\w$]/.test(src[j])) j++; T.push({ t: 'word', v: src.slice(i, j), i }); i = j; continue; }
    82	    if (/[0-9]/.test(c)) { let j = i; while (j < n && /[\w.]/.test(src[j])) j++; T.push({ t: 'num', v: src.slice(i, j), i }); i = j; continue; }
    83	    T.push({ t: 'punct', v: c, i }); i++;
    84	  }
    85	  return T;
    86	}

    91	function grabParen(T, openK) { // openK = index of '(' token
    92	  let depth = 0; const inner = []; let closeTok = null, closeK = -1;
    93	  for (let k = openK; k < T.length; k++) {
    94	    const x = T[k];
    95	    if (x.t === 'punct' && OPEN.has(x.v)) { depth++; if (depth > 1) inner.push(x); continue; }
    96	    if (x.t === 'punct' && CLOSE.has(x.v)) { depth--; if (depth === 0) { closeTok = x; closeK = k; break; } inner.push(x); continue; }
    97	    if (depth >= 1) inner.push(x);
    98	  }
    99	  return { closeTok, closeK, inner };
   100	}

   102	function grabBrace(T, openK) { // openK = index of '{' token; returns index of matching '}' (or -1)
   103	  let depth = 0;
   104	  for (let k = openK; k < T.length; k++) {
   105	    const x = T[k];
   106	    if (x.t === 'punct' && (x.v === '{' || x.v === '${')) depth++;
   107	    else if (x.t === 'punct' && x.v === '}') { depth--; if (depth === 0) return k; }
   108	  }
   109	  return -1;
   110	}

   126	// Find the function whose BODY actually contains token index `idx`. Verifies idx lies between the body
   127	// braces (not merely after a preceding `function` keyword). Returns { name, paramNames, fnStart } | null.
   128	function findEnclosingFunction(T, idx) {
   129	  for (let k = idx - 1; k >= 0; k--) {
   130	    if (!(T[k].t === 'word' && T[k].v === 'function')) continue;
   131	    let p = k + 1, name = null;
   132	    if (T[p] && T[p].t === 'word') { name = T[p].v; p++; }
   133	    if (!(T[p] && T[p].t === 'punct' && T[p].v === '(')) continue;
   134	    const paren = grabParen(T, p);
   135	    if (paren.closeK < 0) continue;
   136	    let b = paren.closeK + 1;
   137	    if (!(T[b] && T[b].t === 'punct' && T[b].v === '{')) continue; // function declarations have a block body right after )
   138	    const bodyClose = grabBrace(T, b);
   139	    if (bodyClose < 0) continue;

   141	      const paramNames = splitArgsTokens(paren.inner).map(g => (g.length === 1 && g[0].t === 'word') ? g[0].v : null).filter(Boolean);
   142	      return { name, paramNames, fnStart: k };
   143	    }
   144	    // a `function` whose body does NOT contain idx is not our encloser; keep scanning outward
   145	  }
   146	  return null;
   147	}
