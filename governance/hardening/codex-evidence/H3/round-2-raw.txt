    45	    const m = key.match(/node_modules\/((?:@[^/]+\/)?[^/]+)$/);
    46	    if (!m) continue;

    52	      // name/version get distinct refs); purl is the ecosystem coordinate (may repeat across nests).
    53	      'bom-ref': key,
    54	      name,
    55	      version,

    58	    };

    85	        { name: 'shippedPath', value: v.shippedPath || '' },
    86	      ],

    98	    version: 1,
    99	    metadata: {

   101	        type: 'application',

   108	        { name: 'generator', value: 'scripts/generate-sbom.js (dependency-free)' },
