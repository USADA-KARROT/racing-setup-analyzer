   154	    if (trainView.finalActivationCheckpoint !== trainSchema.phaseFinalActivationCheckpoint[phase]) fail('TRAIN_PHASE_FINAL_ACTIVATION_MISMATCH', phase, { phase });
   155	
   156	    // ---- forbid R4 capabilities ----
   157	    for (const cap of phaseSchema.capabilities) {
   158	      for (const re of R4_PREFIX_PATTERNS) if (re.test(cap)) fail('R4_CAPABILITY_PRESENT', phase + ' capability ' + cap + ' looks like R4', { phase, capability: cap });
   159	    }
   160	
   161	    // ---- collect checkpoint manifests ----
   162	    const manifestNames = listJson(path.join(phaseDirAbs, 'checkpoints'));
   163	    const manifests = [];
   164	    for (const f of manifestNames) {
   165	      try { manifests.push(readJson(path.join(phaseDirAbs, 'checkpoints', f))); }
   166	      catch (e) { fail('PHASE_CHECKPOINT_MANIFEST_UNREADABLE', phase + '/' + f + ': ' + e.message, { phase, file: f }); }
   167	    }
   168	    phaseManifests[phase] = manifests;
   169	
   170	    phaseSummary[phase] = {
   171	      currentCheckpoint: phaseState.currentCheckpoint,
   172	      finalActivationCheckpoint: trainSchema.phaseFinalActivationCheckpoint[phase],
   173	      authorizedProductionPathCount: Array.isArray(phaseState.authorizedProductionPaths) ? phaseState.authorizedProductionPaths.length : -1,
   174	      enabledCapabilityCount: Array.isArray(phaseState.enabledCapabilities) ? phaseState.enabledCapabilities.length : -1,
   175	      runtimeConsumersAllowed: !!phaseState.runtimeConsumersAllowed,
   176	      uiAllowed: !!phaseState.uiAllowed,
   177	      featureRegistryActivationAllowed: !!phaseState.featureRegistryActivationAllowed,
   178	      algorithmsAllowed: !!phaseState.algorithmsAllowed,
   179	      checkpointManifestCount: manifests.length,
   180	    };
   181	  }
   182	
   183	  // ---- cross-phase advancement rules ----
   184	  const rules = trainSchema.crossPhaseAdvanceRequires || {};
   185	  for (const downstream of Object.keys(rules)) {
   186	    const phaseState = readJsonSafe(path.join(BASE, trainSchema.phaseGovernanceDirs[downstream], 'state.json'));
   187	    const phaseSchema = readJsonSafe(path.join(BASE, trainSchema.phaseGovernanceDirs[downstream], 'schema.json'));
   188	    if (!phaseState || !phaseSchema) continue;
   189	    if (phaseState.currentCheckpoint === phaseSchema.bootstrapCheckpoint) continue; // bootstrap allowed
   190	    const requirement = rules[downstream];
   191	    const [upstream, upstreamCheckpoint] = requirement.split(':');
   192	    const upstreamState = readJsonSafe(path.join(BASE, trainSchema.phaseGovernanceDirs[upstream], 'state.json'));
   193	    if (!upstreamState) { fail('CROSS_PHASE_UPSTREAM_STATE_UNREADABLE', downstream + ' requires ' + upstream + ' state', { phase: downstream }); continue; }
   194	    if (upstreamState.currentCheckpoint !== upstreamCheckpoint) {
   195	      fail('CROSS_PHASE_PREMATURE_ADVANCE', downstream + ' current=' + phaseState.currentCheckpoint + ' but ' + upstream + ' is ' + upstreamState.currentCheckpoint + ' (required ' + upstreamCheckpoint + ')', { downstream, upstream, requiredUpstreamCheckpoint: upstreamCheckpoint, actualUpstreamCheckpoint: upstreamState.currentCheckpoint });
   196	    }
   197	  }
   198	
   199	  // ---- capability + path monotonicity (per phase checkpoint history) ----
   200	  for (const phase of Object.keys(phaseManifests)) {
   201	    const phaseSchema = readJsonSafe(path.join(BASE, trainSchema.phaseGovernanceDirs[phase], 'schema.json'));
   202	    if (!phaseSchema) continue;
   203	    const order = phaseSchema.checkpointOrder;
   204	    const sorted = phaseManifests[phase].slice().filter(m => order.includes(m.checkpoint)).sort((a, b) => indexOf(order, a.checkpoint) - indexOf(order, b.checkpoint));
   205	    let prevCaps = new Set();
   206	    let prevPaths = new Set();
   207	    for (const m of sorted) {
   208	      const after = Array.isArray(m.enabledCapabilitiesAfter) ? m.enabledCapabilitiesAfter : [];
   209	      for (const cap of prevCaps) if (!after.includes(cap)) fail('CAPABILITY_REGRESSION', phase + ' checkpoint ' + m.checkpoint + ' drops capability ' + cap, { phase, checkpoint: m.checkpoint, capability: cap });
   210	      prevCaps = new Set(after);
   211	      const paths = (Array.isArray(m.authorizedPaths) ? m.authorizedPaths : []).map(e => (e && (typeof e === 'string' ? e : e.path)) || null).filter(Boolean);
   212	      for (const p of prevPaths) if (!paths.includes(p)) fail('AUTHORIZED_PATH_REMOVED', phase + ' checkpoint ' + m.checkpoint + ' removed path ' + p, { phase, checkpoint: m.checkpoint, path: p });
   213	      for (const p of paths) prevPaths.add(p);
   214	      // governance change visibility heuristic: if manifest declared changes to governance dirs/files and governanceChanged is false, fail.
   215	      const declaredFiles = collectMaybeFiles(m);
   216	      const touchesGovernance = declaredFiles.some(touchesGovernanceFile);
   217	      if (touchesGovernance && m.governanceChanged !== true) fail('GOVERNANCE_CHANGED_FLAG_MISSING', phase + ' checkpoint ' + m.checkpoint + ' touches governance files but governanceChanged != true', { phase, checkpoint: m.checkpoint });
   218	      // historical lineage: artifactBoundSha (if set) must equal headSha; PASS requires bound SHA match.
   219	      if (m.artifactBoundSha && m.headSha && m.artifactBoundSha !== m.headSha) fail('ARTIFACT_BOUND_SHA_MISMATCH', phase + ' checkpoint ' + m.checkpoint + ' artifactBoundSha=' + m.artifactBoundSha + ' headSha=' + m.headSha, { phase, checkpoint: m.checkpoint });
   220	      if (m.status === 'PASS' && (!m.artifactBoundSha || !m.headSha || m.artifactBoundSha !== m.headSha)) fail('PASS_REQUIRES_ARTIFACT_BOUND_SHA', phase + ' checkpoint ' + m.checkpoint + ' status PASS without SHA-bound artifact', { phase, checkpoint: m.checkpoint });
   221	      // R3.0B case-record schema untouched (E/F only — D has no flag)
   222	      if ((phase === 'R3.0E' || phase === 'R3.0F') && m.r3bCaseRecordSchemaUntouched !== true) fail('R3B_CASE_RECORD_SCHEMA_FLAG_NOT_TRUE', phase + ' checkpoint ' + m.checkpoint + ' r3bCaseRecordSchemaUntouched must be true', { phase, checkpoint: m.checkpoint });
   223	      // R3.0F: package version pin until F6
   224	      if (phase === 'R3.0F' && m.checkpoint !== 'F6_RELEASE' && m.packageVersion !== '1.4.0') fail('R3F_PACKAGE_VERSION_DRIFT', phase + ' checkpoint ' + m.checkpoint + ' packageVersion=' + m.packageVersion + ' (must remain 1.4.0 until F6_RELEASE)', { phase, checkpoint: m.checkpoint });
   225	      if (phase === 'R3.0F' && m.checkpoint === 'F6_RELEASE' && m.packageVersion && m.packageVersion !== '2.0.0' && m.packageVersion !== '1.4.0') fail('R3F_RELEASE_PACKAGE_VERSION_WRONG', 'F6_RELEASE packageVersion must be 1.4.0 (staged) or 2.0.0 (bumped); got ' + m.packageVersion, { phase, checkpoint: m.checkpoint });
   226	      // R3.0F F6 release-mirror consistency: release_executed means PUBLISHED ONLY (semantic
   227	      // ruling recorded in capabilities.json + schema releaseStages). Fail-closed on any
   228	      // record that could pass a tag-only or draft-only state off as an executed release.
   229	      if (phase === 'R3.0F' && m.checkpoint === 'F6_RELEASE') {
   230	        const rr = m.releaseRecord || {};
   231	        const gr = rr.githubRelease || null;
   232	        const stages = (phaseSchema.releaseStages && phaseSchema.releaseStages.order) || [];
   233	        const phaseStateNow = readJsonSafe(path.join(BASE, trainSchema.phaseGovernanceDirs[phase], 'state.json')) || {};
   234	        const capOn = Array.isArray(phaseStateNow.enabledCapabilities) && phaseStateNow.enabledCapabilities.includes('release_executed');
   235	        const isSha = s => typeof s === 'string' && /^[0-9a-f]{40}$/.test(s);
   236	        const published = !!(gr && gr.draft === false && gr.published === true && typeof gr.publishedAt === 'string' && gr.publishedAt.length > 0);
   237	        // stage must come from the schema's monotonic order
   238	        if (m.releaseStage && stages.length && !stages.includes(m.releaseStage)) fail('R3F_RELEASE_STAGE_UNKNOWN', 'F6_RELEASE releaseStage ' + m.releaseStage + ' is not in schema releaseStages.order', { phase, releaseStage: m.releaseStage });
   239	        // release_executed (capability OR record flag) requires an actually-published Release
   240	        if ((capOn || rr.releaseExecuted === true) && (!published || m.releaseStage !== 'RELEASE_PUBLISHED')) fail('R3F_RELEASE_EXECUTED_WITHOUT_PUBLISH', 'release_executed asserted but the GitHub Release is not published (draft/publishedAt/stage mismatch) — a tag alone or a draft Release alone never counts', { phase, capOn, recordFlag: rr.releaseExecuted === true, releaseStage: m.releaseStage, draft: gr && gr.draft, publishedAt: gr && gr.publishedAt });
   241	        // a draft Release on record forbids release_executed in BOTH places
   242	        if (gr && gr.draft === true && (capOn || rr.releaseExecuted === true)) fail('R3F_RELEASE_EXECUTED_ON_DRAFT', 'githubRelease.draft=true but release_executed is asserted', { phase, capOn, recordFlag: rr.releaseExecuted === true });
   243	        // dmgUploaded must corroborate with assets
   244	        const assets = gr && Array.isArray(gr.assets) ? gr.assets : null;
   245	        const hasDmgAsset = !!(assets && assets.some(a => typeof (a && (a.name || a)) === 'string' && /\.dmg$/i.test(a.name || a)));
   246	        if (rr.dmgUploaded === true && !hasDmgAsset) fail('R3F_DMG_FLAG_WITHOUT_ASSET', 'releaseRecord.dmgUploaded=true but githubRelease.assets contains no .dmg', { phase, assets: assets || 'missing' });
   247	        if (gr && assets && assets.length === 0 && rr.dmgUploaded !== false) fail('R3F_DMG_FLAG_NOT_FALSE_ON_EMPTY_ASSETS', 'githubRelease.assets is empty but releaseRecord.dmgUploaded is not exactly false', { phase, dmgUploaded: rr.dmgUploaded });
   248	        // once a tag is recorded: it must be v2.0.0, annotated-with-object, and point exactly at the recorded main merge SHA
   249	        if (rr.tag != null) {
   250	          if (rr.tag !== 'v2.0.0') fail('R3F_RELEASE_TAG_WRONG', 'releaseRecord.tag must be v2.0.0; got ' + rr.tag, { phase, tag: rr.tag });
   251	          if (!isSha(rr.tagTarget) || !isSha(rr.mainMergeSha) || rr.tagTarget !== rr.mainMergeSha) fail('R3F_TAG_TARGET_MISMATCH', 'releaseRecord.tagTarget must be the exact recorded mainMergeSha (both 40-hex)', { phase, tagTarget: rr.tagTarget, mainMergeSha: rr.mainMergeSha });
   252	          if (rr.tagType !== 'annotated' || !isSha(rr.tagObjectSha) || rr.tagObjectSha === rr.tagTarget) fail('R3F_TAG_NOT_ANNOTATED', 'recorded tag must be annotated with its own tag-object SHA distinct from the peeled commit', { phase, tagType: rr.tagType, tagObjectSha: rr.tagObjectSha });
   253	        }
   254	        if (gr && gr.tagName != null && gr.tagName !== rr.tag) fail('R3F_RELEASE_TAG_NAME_MISMATCH', 'githubRelease.tagName differs from releaseRecord.tag', { phase, tagName: gr.tagName, tag: rr.tag });
   255	        // stage/evidence coherence (monotonic, no history rewrite)
   256	        if (m.releaseStage === 'RELEASE_TAGGED_DRAFT' && rr.tag == null && !gr) fail('R3F_RELEASE_STAGE_EVIDENCE_MISMATCH', 'releaseStage RELEASE_TAGGED_DRAFT but neither a tag nor a githubRelease is recorded', { phase });
   257	        if (m.releaseStage === 'RELEASE_PREPARED' && (rr.tag != null || gr)) fail('R3F_RELEASE_STAGE_EVIDENCE_MISMATCH', 'releaseStage RELEASE_PREPARED but tag/Release evidence already recorded (stage may not lag evidence)', { phase });
   258	        if (m.releaseStage === 'RELEASE_PUBLISHED' && !published) fail('R3F_RELEASE_STAGE_EVIDENCE_MISMATCH', 'releaseStage RELEASE_PUBLISHED but githubRelease is not published', { phase });
   259	      }
   260	    }
   261	  }
   262	
   263	  // ---- production guard: no R3.0B frozen module modification declared ----
   264	  // (Static check: train validator just asserts that none of the per-checkpoint manifests declare any of
   265	  //  the frozen R3.0B paths as authorizedPaths. The frozen-boundary check separately catches actual diff.)
   266	  for (const phase of Object.keys(phaseManifests)) {
   267	    for (const m of phaseManifests[phase]) {
   268	      const paths = (Array.isArray(m.authorizedPaths) ? m.authorizedPaths : []).map(e => (e && (typeof e === 'string' ? e : e.path)) || null).filter(Boolean);
   269	      for (const p of paths) if (FROZEN_R3B_PATHS.includes(p)) fail('AUTH_PATH_TOUCHES_R3B_FROZEN', phase + ' checkpoint ' + m.checkpoint + ' authorizes frozen R3.0B path ' + p, { phase, checkpoint: m.checkpoint, path: p });
   270	    }
   271	  }
   272	
   273	  return finish({
   274	    ok: violations.length === 0,
   275	    violations,
   276	    trainStatus: train.trainStatus,
   277	    currentPhase: train.currentPhase,
   278	    currentPhaseCheckpoint: train.currentPhaseCheckpoint,
   279	    targetVersion: train.targetVersion,
   280	    targetTag: train.targetTag,
   281	    intermediateReleaseAllowed: train.intermediateReleaseAllowed,
   282	    r4Excluded: train.r4Excluded,
   283	    phases: phaseSummary,
   284	    historicalCheckpointsByPhase: Object.keys(phaseManifests).reduce(function (acc, p) { acc[p] = phaseManifests[p].map(function (m) { return { checkpoint: m.checkpoint, headSha: m.headSha, status: m.status, governanceChanged: !!m.governanceChanged }; }); return acc; }, {}),
   285	  });
   286	}
